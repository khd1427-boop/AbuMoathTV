# Builds the ABU MOATH TV app automatically every time files are uploaded to GitHub.
# The finished app is published under "Releases" with a fixed download link:
#   https://github.com/<your-username>/<repo-name>/releases/latest/download/AbuMoathTV.apk
# Type that link in the "Downloader" app on the TV to install or update.
name: Build APK

on:
  push:
    branches: [main, master]
  workflow_dispatch:

permissions:
  contents: write

jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4

      - uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: "17"

      - uses: android-actions/setup-android@v3

      - uses: gradle/actions/setup-gradle@v4

      - name: Unit tests
        run: chmod +x gradlew && ./gradlew --no-daemon testDebugUnitTest

      - name: Build APK
        run: ./gradlew --no-daemon assembleRelease -PversionCodeOverride=${{ github.run_number }}

      - name: Publish release
        uses: softprops/action-gh-release@v2
        with:
          tag_name: build-${{ github.run_number }}
          name: "ABU MOATH TV – إصدار ${{ github.run_number }}"
          body: "حمّل التطبيق على التلفاز عبر Downloader من الرابط: https://github.com/${{ github.repository }}/releases/latest/download/AbuMoathTV.apk"
          make_latest: "true"
          files: app/build/outputs/apk/release/AbuMoathTV.apk
